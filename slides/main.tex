\begin{frame}[t]{Version control}
  \practicehead{code-branch}{5}{Every change recorded, every mistake undoable.}
  \begin{columns}[t]
    \column{0.54\textwidth}\vspace{0pt}
    \begin{itemize}
      \item See what changed, when and why.
      \item Backup and collaboration in one.
      \item Branches let features develop in parallel.
      \item Commit before an agent edits, then review its diff.
    \end{itemize}

    \column{0.42\textwidth}\vspace{0pt}
    \begin{toolpanel}
      \toolrow{Any}{\pick{git}}
      \toolrow{Hosting}{\pick{GitHub} \opt{GitLab}}
      \toolrow{Interfaces}{\opt{VS Code} \opt{GitHub Desktop} \opt{GitHub CLI (gh)}}
    \end{toolpanel}
  \end{columns}
  \practicetracker{1}
  \note[item]{Commit small and often, with messages that say why.}
  \note[item]{Git worktrees are extra checkouts of the repo on other branches, so several
  agent sessions can work at once (part 2).}
\end{frame}

\begin{frame}[t]{Environments and package management}
  \practicehead{boxes}{5}{One environment per project, pinned by a lockfile.}
  \begin{columns}[t]
    \column{0.54\textwidth}\vspace{0pt}
    \begin{itemize}
      \item Code depends on package versions (e.g.\ numpy~2).
      \item A shared global environment drifts and breaks old projects.
      \item A lockfile (exact versions of every package) lets anyone, and the cluster,
        rebuild it.
    \end{itemize}

    \column{0.42\textwidth}\vspace{0pt}
    \begin{toolpanel}
      \toolrow{Python}{\pick{pixi} \opt{uv} \opt{conda/mamba} \opt{pip + venv}}
      \toolrow{R}{\pick{renv} \opt{pak} \opt{rig}}
      \toolrow{Julia}{\opt{Pkg}}
      \toolrow{Containers}{\opt{Docker} \opt{Apptainer}}
    \end{toolpanel}
  \end{columns}
  \practicetracker{2}
  \note[item]{pixi installs from conda-forge and PyPI, with a cross-platform lockfile and
  tasks; it handles R and non-Python dependencies too. uv is excellent if everything is on
  PyPI. rig manages R versions.}
\end{frame}

\begin{frame}[t]{Testing}
  \practicehead{vial}{4}{How you know the code, yours or the AI's, is right.}
  \begin{columns}[t]
    \column{0.54\textwidth}\vspace{0pt}
    \begin{itemize}
      \item Test against known answers: analytic results, limiting cases,
        simulate-and-recover.
      \item Stochastic code: pass a seeded \texttt{rng} in; compare with tolerances.
      \item Write the failing test first (TDD): ideal with agents.
      \item Then refactor knowing nothing broke.
    \end{itemize}

    \column{0.42\textwidth}\vspace{0pt}
    \begin{toolpanel}
      \toolrow{Python}{\pick{pytest} \opt{hypothesis} \opt{pytest-cov}}
      \toolrow{R}{\pick{testthat}}
      \toolrow{Julia}{\opt{Test}}
    \end{toolpanel}
  \end{columns}
  \practicetracker{3}
  \note[item]{Simulate-and-recover: simulate data from known parameters and check that the
  inference recovers them.}
  \note[item]{Stochastic code: pass \texttt{rng = np.random.default\_rng(seed)} into functions
  rather than setting a global seed; test with \texttt{assert\_allclose} or statistical
  tolerances, not exact values.}
  \note[item]{With an agent: write or approve the test yourself, then ask it to make the test
  pass without editing it. hypothesis does property-based testing; pytest-cov measures
  coverage.}
\end{frame}

\begin{frame}[t]{Archiving code}
  \practicehead{archive}{4}{A permanent copy of the code behind each paper.}
  \begin{columns}[t]
    \column{0.54\textwidth}\vspace{0pt}
    \begin{itemize}
      \item Archive the exact version used for the paper.
      \item A DOI keeps the paper's link permanent, even if the repo moves.
      \item Zenodo's GitHub integration needs a public repo; otherwise upload the release by
        hand.
    \end{itemize}

    \column{0.42\textwidth}\vspace{0pt}
    \begin{toolpanel}
      \toolrow{Archives}{\pick{Zenodo} \opt{Software Heritage} \opt{OSF}}
    \end{toolpanel}
  \end{columns}
  \practicetracker{4}
  \note[item]{Zenodo archives each GitHub release automatically once the repo is public, and
  gives a citation if one is needed.}
\end{frame}

\begin{frame}[t]{Formatting and linting}
  \practicehead{broom}{3}{Consistent, bug-checked code for free.}
  \begin{columns}[t]
    \column{0.54\textwidth}\vspace{0pt}
    \begin{itemize}
      \item Consistent style, with no effort or arguments.
      \item Linters catch real bugs, e.g.\ undefined names, unused imports.
      \item Runs on save, or after each agent edit (a hook: a command run automatically).
    \end{itemize}

    \column{0.42\textwidth}\vspace{0pt}
    \begin{toolpanel}
      \toolrow{Python}{\pick{ruff} \opt{pylint}}
      \toolrow{R}{\opt{air} \opt{styler} \opt{lintr}}
      \toolrow{Julia}{\opt{JuliaFormatter.jl} \opt{Runic.jl}}
      \toolrow{Any}{\opt{pre-commit}}
    \end{toolpanel}
  \end{columns}
  \practicetracker{5}
  \note[item]{ruff replaces black, isort and flake8 in one fast tool. pre-commit runs checks
  before each commit.}
\end{frame}

\begin{frame}[t]{Workflow management}
  \practicehead{project-diagram}{3}{One command rebuilds every result.}
  \begin{columns}[t]
    \column{0.54\textwidth}\vspace{0pt}
    \begin{itemize}
      \item Reruns only what changed when code or data change.
      \item Makes the dependencies between data, code and outputs explicit.
      \item Scales to the cluster; matters more as projects grow.
    \end{itemize}

    \column{0.42\textwidth}\vspace{0pt}
    \begin{toolpanel}
      \toolrow{Any}{\pick{Snakemake} \opt{Nextflow} \opt{Make}}
      \toolrow{R}{\opt{targets}}
      \toolrow{Simple}{\opt{pixi tasks}}
    \end{toolpanel}
  \end{columns}
  \practicetracker{6}
  \note[item]{Snakemake works with any language, Julia included: each rule just runs a
  command.}
\end{frame}

\begin{frame}[t]{Editor and debugger}
  \practicehead{laptop-code}{3}{See errors as you type; inspect, don't print.}
  \begin{columns}[t]
    \column{0.54\textwidth}\vspace{0pt}
    \begin{itemize}
      \item The editor flags errors and makes code easy to navigate.
      \item A debugger pauses the code so you can inspect its state.
      \item Notebooks for exploring; scripts once it matters.
    \end{itemize}

    \column{0.42\textwidth}\vspace{0pt}
    \begin{toolpanel}
      \toolrow{Editors}{\pick{VS Code} \opt{PyCharm} \opt{Positron} \opt{RStudio}}
      \toolrow{Debuggers}{\pick{VS Code debugger} \opt{pdb} \opt{R browser()}
        \opt{Infiltrator.jl}}
    \end{toolpanel}
  \end{columns}
  \practicetracker{7}
  \note[item]{Notebooks hide state and let cells run out of order; move code to scripts or
  the package once results depend on it.}
\end{frame}

\begin{frame}[t]{Type checking}
  \practicehead{spell-check}{2}{Catch wrong-type bugs before running.}

      \item Type hints document interfaces, and help editors and AI.
      \item Lower priority for exploratory code, but cheap to add with AI.

  \end{columns}

\begin{frame}{Organising code}
  \vspace{3mm}
  \centering
  \begin{tikzpicture}[file/.style={stepbox, draw=oxgrey, dashed, font=\footnotesize\ttfamily,
    minimum height=8mm},
    scr/.style={stepbox, fill=oxtint, font=\footnotesize}]
    \node[scr] (an) {\texttt{final\_size.py}\\{\scriptsize\color{oxgrey}compute, save}};
    \node[file, right=7mm of an] (res) {results/};
    \node[scr, right=7mm of res] (pl) {\texttt{final\_size\_plots.py}\\{\scriptsize\color{oxgrey}load, plot}};
    \node[file, right=7mm of pl] (fig) {figures/};
    \draw[arr] (an) -- (res); \draw[arr] (res) -- (pl); \draw[arr] (pl) -- (fig);
    \node[stepbox, fill=oxfordblue, text=white, minimum height=9mm, anchor=south,
      minimum width=60mm] (pkg) at ($(an.north)!0.5!(pl.north)+(0,9mm)$)
    {\texttt{src/mypkg/}\quad models, likelihoods, helpers};
    \draw[arr] (pkg.south -| an.north) -- node[right, font=\scriptsize, text=oxgrey]{import} (an.north);
    \draw[arr] (pkg.south -| pl.north) -- node[right, font=\scriptsize, text=oxgrey]{import} (pl.north);
  \end{tikzpicture}\par
  \vspace{2mm}
  \raggedright
  \begin{itemize}
    \item \textbf{Don't repeat yourself:} define each value once, e.g.\ the serial interval.
      A repeated one-liner is fine.
    \item \textbf{One function per task:} \texttt{simulate\_outbreak}, \texttt{plot\_risk}.
      Pass seeds in, no hidden globals.
    \item \textbf{Public vs private:} public functions are documented and tested;
      \texttt{\_helpers} can change freely.
  \end{itemize}
  \note[item]{Separating analysis from plotting means figures can be tweaked without
  recomputing.}
\end{frame}

\begin{frame}{Documenting code}
  \begin{itemize}
    \item README: what it is, how to install it, how to reproduce the results.
    \item Docstrings for public functions: purpose, arguments, returns.
    \item Comments say \emph{why}, not \emph{what}.
    \item Names say what things are, with units (\texttt{incubation\_days}).
    \item No commented-out code: git remembers it.
    \item Doc websites only for shared packages: \pick{Sphinx} \opt{roxygen2 + pkgdown}
      \opt{Documenter.jl}
  \end{itemize}
  \note[item]{Docstrings in Google or NumPy style.}
\end{frame}

\begin{frame}{Risks: correctness and skills}
  \begin{tcbraster}[raster columns=2, raster equal height=rows, raster column skip=3mm,
    raster row skip=3mm]
    \begin{riskcard}{Plausible but wrong}
      \eg{Runs and looks fine, but e.g.\ gamma scale vs rate, or an off-by-one in a
      renewal-equation convolution.}
      \fix Check against analytic results, limiting cases, simulate-and-recover.
    \end{riskcard}
    \begin{riskcard}{Tests that pass for the wrong reason}
      \eg{Tests that mirror the code, or get edited or skipped to pass.}
      \fix Decide what the tests check, and read them.
    \end{riskcard}
    \begin{riskcard}{Bloat and hidden fallbacks}
      \eg{More code than needed; fallbacks that hide errors.}
      \fix Small steps, read every diff, ask for simpler code.
    \end{riskcard}
    \begin{riskcard}{Losing understanding}
      \eg{You can't defend code you don't understand.}
      \fix Ask it to explain; write some code yourself.
    \end{riskcard}
  \end{tcbraster}
  \note[item]{Gamma: numpy's \texttt{random.gamma} and scipy's \texttt{stats.gamma} take a
  scale; R's \texttt{rgamma} takes a rate by default; papers quote either. A classic
  translation bug.}
\end{frame}

\begin{frame}{Risks: data, security and policy}
  \begin{tcbraster}[raster columns=2, raster equal height=rows, raster column skip=3mm,
    raster row skip=3mm]
    \begin{riskcard}{Confidential or patient-level data}
      \eg{Everything the agent reads or prints goes to the provider.}
      \fix Use University-approved tools (ChatGPT Edu, Codex; Claude under discussion);
      check data agreements; develop on synthetic data.
    \end{riskcard}
    \begin{riskcard}{Destructive commands}
      \eg{Depending on the mode, agents run commands without asking.}
      \fix An ask-first mode or a sandbox for sensitive work; commit first.
    \end{riskcard}
    \begin{riskcard}{Invented packages, hidden instructions}
      \eg{Packages that don't exist (or are malicious); instructions hidden in untrusted
      code or web pages.}
      \fix Check new dependencies in the lockfile diff; take care with unfamiliar repos.
    \end{riskcard}
    \begin{riskcard}{Journal, funder and thesis rules}
      \eg{AI can't be an author; policies vary.}
      \fix Check them, and say how AI was used.
    \end{riskcard}
  \end{tcbraster}
